// ========================== // Load Orders (優化防錯版) // ========================== async function loadOrders(){ if(!currentUser){ document.getElementById("orderHistory").innerHTML = "Login to view orders."; return; } try { const q = query( collection(db,"orders"), where("userId","==",currentUser.uid), orderBy("orderDate","desc") ); const snapshot = await getDocs(q); let html = ""; snapshot.forEach(doc=>{ const o = doc.data(); const itemDetails = (o.items || []) .map(item => `${item.name} x${item.quantity} (NT$${item.price})`) .join("
"); html += `
Items:
${itemDetails}
Customer Name: ${o.customerName || 'N/A'}
Customer Email: ${o.customerEmail || 'N/A'}
Total Amount: NT$${o.total}
Status: ${o.status || 'Pending'}
Account Email: ${o.userEmail}
`; }); if(html===""){ html="No orders found."; } document.getElementById("orderHistory").innerHTML = html; } catch (error) { console.error("Error loading orders:", error); // 🆕 自動偵測是否為 Index 錯誤，如果是，直接把後台連結印在網頁畫面上 if (error.message && error.message.includes("FAILED_PRECONDITION")) { document.getElementById("orderHistory").innerHTML = `
⚠️ 資料庫索引尚未建立：
請開啟瀏覽器 F12 檢查 Console 主控台，點擊紅字錯誤中的 Firebase 官方連結來自動建立索引。
(建立後需等待約 3 分鐘方可正常顯示訂單)
`; } else { document.getElementById("orderHistory").innerHTML = "Error loading orders: " + error.message; } } }
